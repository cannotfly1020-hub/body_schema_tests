const CACHE_NAME = 'brain-gps-dock-v1'; const ASSETS_TO_CACHE = [ './index.html', './manifest.json' ]; self.addEventListener('install', (event) => { event.waitUntil( caches.open(CACHE_NAME).then((cache) => { return cache.addAll(ASSETS_TO_CACHE); }).then(() => self.skipWaiting()) ); }); self.addEventListener('activate', (event) => { event.waitUntil( caches.keys().then((keys) => { return Promise.all( keys.map((key) => { if (key !== CACHE_NAME) { return caches.delete(key); } }) ); }).then(() => self.clients.claim()) ); }); self.addEventListener('fetch', (event) => { // Cache-first falling back to network strategy for local app assets if (event.request.method !== 'GET') return; event.respondWith( caches.match(event.request).then((cachedResponse) => { if (cachedResponse) { return cachedResponse; } return fetch(event.request).then((networkResponse) => { return networkResponse; }).catch(() => { // Fallback or ignore for CDN calls }); }) ); });
