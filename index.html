import React, { useEffect, useRef, useState, useCallback } from 'react'; import { Pose, Results, POSE_CONNECTIONS } from '@mediapipe/pose'; import { Camera } from '@mediapipe/camera_utils'; // ========================================== // Utils: バイオメカニクス演算 & 音声合成 // ========================================== const calcAngle = (p1: any, p2: any) => { return Math.atan2(p2.y - p1.y, p2.x - p1.x) * (180 / Math.PI); }; const speak = (text: string) => { const ut = new SpeechSynthesisUtterance(text); ut.lang = 'ja-JP'; ut.rate = 1.2; window.speechSynthesis.speak(ut); }; const playTone = (ctx: AudioContext, freq: number, type: OscillatorType = 'sine', duration = 0.2) => { const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = type; osc.frequency.setValueAtTime(freq, ctx.currentTime); gain.gain.setValueAtTime(0.5, ctx.currentTime); gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + duration); osc.connect(gain); gain.connect(ctx.destination); osc.start(); osc.stop(ctx.currentTime + duration); }; // ========================================== // Types & Constants // ========================================== type AppState = 'SETUP' | 'WAITING_FOR_BODY' | 'COUNTDOWN' | 'RUNNING' | 'DRILL_SUCCESS' | 'REPLAY'; type TestType = 'BLINDFOLD_STEP' | 'T_POSE' | 'SINGLE_LEG' | 'X_FACTOR'; const REQUIRED_LANDMARKS = [0, 11, 12, 23, 24, 25, 26, 27, 28, 29, 30]; // 鼻, 肩, 腰, 膝, 足首/踵 // ========================================== // Main Component // ========================================== export default function App() { const videoRef = useRef(null); const canvasRef = useRef(null); const [appState, setAppState] = useState('SETUP'); const [testType, setTestType] = useState('T_POSE'); // 状態管理 const [awakeningScore, setAwakeningScore] = useState(0); const [stableFrames, setStableFrames] = useState(0); const [drillReps, setDrillReps] = useState(0); // Audio Context (ユーザーインタラクション後に初期化) const audioCtxRef = useRef(null); // Recorder const mediaRecorderRef = useRef(null); const recordedChunksRef = useRef([]); const [replayUrl, setReplayUrl] = useState(null); const startTimeRef = useRef(0); // ------------------------------------------ // Core: 姿勢推定コールバック // ------------------------------------------ const onResults = useCallback((results: Results) => { if (!canvasRef.current || !results.poseLandmarks) return; const ctx = canvasRef.current.getContext('2d'); if (!ctx) return; ctx.save(); ctx.clearRect(0, 0, canvasRef.current.width, canvasRef.current.height); // カメラ映像の描画 ctx.drawImage(results.image, 0, 0, canvasRef.current.width, canvasRef.current.height); const lm = results.poseLandmarks; // 1. 全身認識＆フレームアウト検知ロジック let isFullBodyVisible = true; let isFrameOut = false; REQUIRED_LANDMARKS.forEach(idx => { const pt = lm[idx]; if (pt.visibility === undefined || pt.visibility < 0.6) isFullBodyVisible = false; // マージン5%でのフレームアウト判定 if (pt.x < 0.05 || pt.x > 0.95 || pt.y < 0.05 || pt.y > 0.95) isFrameOut = true; }); // 2. 状態遷移ロジック setAppState((current) => { if (current === 'WAITING_FOR_BODY') { if (isFullBodyVisible && !isFrameOut) { setStableFrames(prev => { const next = prev + 1; if (next === 10) { speak("骨格をロックしました。3秒後にスタート"); return next; } if (next > 30) return 30; // カウントダウントリガー後 return next; }); if (stableFrames === 29) return 'COUNTDOWN'; // 簡易的に次フレームで遷移 } else { setStableFrames(0); } } if (current === 'RUNNING') { const timeElapsed = performance.now() - startTimeRef.current; // 1.8秒以降のフレームアウトで即時終了 (ブザー) if (timeElapsed > 1800 && isFrameOut) { if (audioCtxRef.current) playTone(audioCtxRef.current, 150, 'sawtooth', 0.5); speak("エリア外です！リプレイを確認しましょう"); stopRecording(); return 'REPLAY'; } // T-Pose (肩甲骨水平ドリル) のリアルタイム判定例 if (testType === 'T_POSE') { const lShoulder = lm[11], rShoulder = lm[12], lWrist = lm[15], rWrist = lm[16]; const lAngle = Math.abs(calcAngle(lShoulder, lWrist)); // 理想は 180 or 0 const rAngle = Math.abs(calcAngle(rShoulder, rWrist)); // 左右差と水平度（覚醒度）の算出 const accuracy = Math.max(0, 100 - (Math.abs(180 - lAngle) + Math.abs(0 - rAngle)) * 2); setAwakeningScore(Math.floor(accuracy)); // 覚醒度90%以上でエメラルドグリーンに ctx.strokeStyle = accuracy > 90 ? '#10B981' : '#38BDF8'; ctx.lineWidth = 6; if (accuracy > 90) { // ここで3秒キープのロジックを入れる（今回は即時カウント） setDrillReps(r => { if (r === 2) { speak("覚醒完了！素晴らしいアハ体験です！"); stopRecording(); return 3; } return r + 1; }); } } } return current; }); // 骨格の描画 (MediaPipeユーティリティを使わず手動描画でパフォーマンス・デザイン調整) if (appState === 'RUNNING' || appState === 'WAITING_FOR_BODY') { ctx.strokeStyle = (appState === 'RUNNING' && awakeningScore > 90) ? '#10B981' : '#60A5FA'; ctx.lineWidth = 4; POSE_CONNECTIONS.forEach(([i, j]) => { if (lm[i].visibility! > 0.5 && lm[j].visibility! > 0.5) { ctx.beginPath(); ctx.moveTo(lm[i].x * canvasRef.current!.width, lm[i].y * canvasRef.current!.height); ctx.lineTo(lm[j].x * canvasRef.current!.width, lm[j].y * canvasRef.current!.height); ctx.stroke(); } }); // ランドマークの描画 ctx.fillStyle = '#FBBF24'; // Brawl Stars Amber lm.forEach((pt) => { if (pt.visibility! > 0.5) { ctx.beginPath(); ctx.arc(pt.x * canvasRef.current!.width, pt.y * canvasRef.current!.height, 4, 0, 2 * Math.PI); ctx.fill(); } }); } ctx.restore(); }, [appState, stableFrames, testType, awakeningScore]); // ------------------------------------------ // Camera & MediaPipe Init // ------------------------------------------ useEffect(() => { const pose = new Pose({ locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/pose/${file}`, }); pose.setOptions({ modelComplexity: 1, smoothLandmarks: true, minDetectionConfidence: 0.7, minTrackingConfidence: 0.7, }); pose.onResults(onResults); if (videoRef.current) { const camera = new Camera(videoRef.current, { onFrame: async () => { await pose.send({ image: videoRef.current! }); }, width: 640, height: 480, facingMode: 'user' }); camera.start(); } }, [onResults]); // ------------------------------------------ // Game Sequence Control // ------------------------------------------ const startGame = () => { audioCtxRef.current = new AudioContext(); // ユーザー操作でAudio解放 speak("テストを開始します。枠内に全身を入れてください。"); setAppState('WAITING_FOR_BODY'); setStableFrames(0); setAwakeningScore(0); setDrillReps(0); }; useEffect(() => { if (appState === 'COUNTDOWN') { let count = 3; const iv = setInterval(() => { if (count > 0) { if (audioCtxRef.current) playTone(audioCtxRef.current, 440, 'sine', 0.1); count--; } else { clearInterval(iv); if (audioCtxRef.current) playTone(audioCtxRef.current, 880, 'sine', 0.3); setAppState('RUNNING'); startRecording(); startTimeRef.current = performance.now(); } }, 1000); return () => clearInterval(iv); } }, [appState]); // ------------------------------------------ // Recording (MediaRecorder) // ------------------------------------------ const startRecording = () => { if (!canvasRef.current) return; const stream = canvasRef.current.captureStream(30); // 30 FPS mediaRecorderRef.current = new MediaRecorder(stream, { mimeType: 'video/webm' }); mediaRecorderRef.current.ondataavailable = (e) => { if (e.data.size > 0) recordedChunksRef.current.push(e.data); }; mediaRecorderRef.current.onstop = () => { const blob = new Blob(recordedChunksRef.current, { type: 'video/webm' }); setReplayUrl(URL.createObjectURL(blob)); recordedChunksRef.current = []; // リセット setAppState('REPLAY'); }; mediaRecorderRef.current.start(); }; const stopRecording = () => { if (mediaRecorderRef.current && mediaRecorderRef.current.state === 'recording') { mediaRecorderRef.current.stop(); } }; const closeReplay = () => { if (replayUrl) URL.revokeObjectURL(replayUrl); setReplayUrl(null); setAppState('SETUP'); }; // ------------------------------------------ // UI Render (Brawl Stars Style) // ------------------------------------------ return (
{/* 隠しVideo要素 */}
); }
